# Desktop shell and independently distributable numerical core.
cmake_minimum_required(VERSION 3.24)
project(SLXOrbitLab VERSION 0.7.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SLX_BUILD_DESKTOP "Build the Qt mission console" ON)
set(_private_default OFF)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/private/core/mission.cpp")
    set(_private_default ON)
endif()
option(SLX_BUILD_PRIVATE_CORE "Build the Silicon LogiX implementation" ${_private_default})
if(WIN32)
    set(_sdk_default "${CMAKE_CURRENT_SOURCE_DIR}/sdk/windows-mingw/SiliconLogiX.OrbitLab.Core.dll.a")
else()
    set(_sdk_default "${CMAKE_CURRENT_SOURCE_DIR}/sdk/linux-x86_64/SiliconLogiX.OrbitLab.Core.so")
endif()
set(SLX_CORE_BINARY "${_sdk_default}" CACHE FILEPATH "ABI-compatible core library")
if(SLX_BUILD_PRIVATE_CORE)
    add_library(slx-orbit-core SHARED private/core/mission.cpp)
    target_compile_definitions(slx-orbit-core PRIVATE SLX_CORE_BUILD)
    target_include_directories(slx-orbit-core PUBLIC core/include)
    set_target_properties(slx-orbit-core PROPERTIES OUTPUT_NAME "SiliconLogiX.OrbitLab.Core" PREFIX "" IMPORT_PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
    if(WIN32)
        configure_file(assets/brand/core-version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/core-version.rc" @ONLY)
        target_sources(slx-orbit-core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/core-version.rc")
    endif()
else()
    if(NOT EXISTS "${SLX_CORE_BINARY}")
        message(FATAL_ERROR "Set SLX_CORE_BINARY to the approved core import library")
    endif()
    add_library(slx-orbit-core SHARED IMPORTED)
    target_include_directories(slx-orbit-core INTERFACE core/include)
    if(WIN32)
        get_filename_component(_sdk_directory "${SLX_CORE_BINARY}" DIRECTORY)
        set_target_properties(slx-orbit-core PROPERTIES IMPORTED_IMPLIB "${SLX_CORE_BINARY}" IMPORTED_LOCATION "${_sdk_directory}/SiliconLogiX.OrbitLab.Core.dll")
    else()
        set_target_properties(slx-orbit-core PROPERTIES IMPORTED_LOCATION "${SLX_CORE_BINARY}")
    endif()
endif()
if(SLX_BUILD_DESKTOP)
    set(CMAKE_AUTOMOC ON)
    set(CMAKE_AUTORCC ON)
    find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network Svg Test)
    qt_add_executable(slx-orbitlab src/main.cpp src/controller.cpp src/controller.h src/telemetry_model.cpp src/telemetry_model.h src/godot_inspector.cpp src/godot_inspector.h src/desktop_check.cpp src/desktop_check.h resources.qrc)
    target_link_libraries(slx-orbitlab PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Svg Qt6::Test)
    target_compile_definitions(slx-orbitlab PRIVATE SLX_APP_VERSION="${PROJECT_VERSION}")
    set_target_properties(slx-orbitlab PROPERTIES WIN32_EXECUTABLE TRUE)
    if(WIN32)
        qt_add_executable(slx-icon-maker EXCLUDE_FROM_ALL scripts/make-icon.cpp)
        target_link_libraries(slx-icon-maker PRIVATE Qt6::Gui Qt6::Svg)
        if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/brand/orbitlab-icon.ico")
            configure_file(assets/brand/app-version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app-version.rc" @ONLY)
            target_sources(slx-orbitlab PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app-version.rc")
        endif()
    endif()
endif()
include(CTest)
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/core_tests.cpp")
    add_executable(core-tests tests/core_tests.cpp)
    target_link_libraries(core-tests PRIVATE slx-orbit-core)
    if(WIN32 AND NOT SLX_BUILD_PRIVATE_CORE)
        add_custom_command(TARGET core-tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:slx-orbit-core>" "$<TARGET_FILE_DIR:core-tests>")
    endif()
    add_test(NAME mission-core-contract COMMAND core-tests)
endif()
if(BUILD_TESTING AND SLX_BUILD_DESKTOP)
    add_test(NAME desktop-startup COMMAND slx-orbitlab --smoke-test --no-backend-start)
    set_tests_properties(desktop-startup PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_QPA_FONTDIR=$ENV{SystemRoot}/Fonts")
    if(WIN32)
        find_package(Python3 COMPONENTS Interpreter QUIET)
        if(Python3_FOUND)
            add_test(NAME desktop-runtime COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/check-desktop-runtime.py" --application "$<TARGET_FILE:slx-orbitlab>")
        endif()
    endif()
endif()
